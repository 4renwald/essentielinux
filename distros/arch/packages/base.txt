# Arch base packages. Lines: `*name :: description` — `*` marks a structural
# package that cannot be deselected (networking, hardware probing, swap, and
# the settings backend the desktop's theming depends on).
*networkmanager :: network management
*pciutils :: lspci hardware probing
*zram-generator :: compressed swap on zram
*xdg-user-dirs :: manages ~/Documents, ~/Downloads and friends
*xdg-utils :: xdg-open and desktop integration helpers
*dconf :: GSettings backend required by caelestia scheme theming
curl :: transfers and downloads
git :: version control
btop :: terminal resource monitor
unzip :: zip extraction
bluez :: Bluetooth stack
bluez-utils :: Bluetooth utilities
linux-firmware :: firmware for hardware components
wireless-regdb :: regulatory database for full 5/6 GHz Wi-Fi
noto-fonts :: Noto font family
noto-fonts-emoji :: color emoji font
ttf-dejavu :: DejaVu fonts
ttf-nerd-fonts-symbols :: Nerd Font symbols
ttf-jetbrains-mono-nerd :: font Caelestia asks for by name
man-db :: manual page reader
man-pages :: Linux manual pages
less :: pager used by man and git
nano :: console editor for recovery sessions
noto-fonts-cjk :: CJK fonts so web text does not render as boxes
ttf-liberation :: metric-compatible stand-ins for Arial/Times/Courier
pacman-contrib :: paccache cache pruning plus checkupdates and pactree
reflector :: mirrorlist ranking
fwupd :: firmware updates over LVFS
smartmontools :: drive health monitoring (SMART)
